## CaptureModels.kt
data class CaptureCapabilities(
  val nativeCaptureSupported: Boolean,
  val apiLevel: Int,
  val minNativeApi: Int = CaptureRuntime.MIN_NATIVE_API,
enum class CaptureStatus
  Ok
  SkippedByDHash
  UnsupportedApi
  UnsupportedRenderMode
  SurfaceUnavailable
  Timeout
  PixelCopyFailed
  ProcessingFailed
  Cancelled
  Disposed
enum class CaptureCoverage
  EngineSurface
enum class RenderMode
  SurfaceView
  TextureView
  Hybrid
  Unknown
data class NormalizedMask(
  val x: Double,
  val y: Double,
  val width: Double,
  val height: Double,
data class CaptureRequest(
  val requestId: Long,
  val pixelWidth: Int,
  val pixelHeight: Int,
  val force: Boolean,
  val lastDHash: String = "",
  val masks: List<NormalizedMask> = emptyList(),
data class CaptureTimings(
  val surfaceCopyMicros: Long = 0,
  val maskFillMicros: Long = 0,
  val dHashMicros: Long = 0,
  val jpegMicros: Long = 0,
  val sha256Micros: Long = 0,
  val pixelReadbackMicros: Long = 0,
data class CaptureResult(
  val requestId: Long,
  val status: CaptureStatus,
  val coverage: CaptureCoverage? = null,
  val jpeg: ByteArray = ByteArray(0),
  val width: Int = 0,
  val height: Int = 0,
  val dHash: String = "",
  val contentHash: String = "",
  val timings: CaptureTimings = CaptureTimings(),
  val renderMode: RenderMode = RenderMode.Unknown,
  val incomplete: Boolean = false,

## CaptureRuntime.kt
class CaptureRuntime(
  fun capabilities(): CaptureCapabilities
  fun capture(view: View, request: CaptureRequest, onComplete: (CaptureResult) -> Unit)
  fun cancel(requestId: Long)
  fun dispose()
  companion object
  const val MIN_NATIVE_API: Int = 24
  const val DEFAULT_TIMEOUT_MS: Long = 2_000
  const val JPEG_QUALITY: Int = 80
